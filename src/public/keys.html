For Agents (Programmatic)

# Step 1: Get challenge
CHALLENGE=$(curl -s -X POST https://skillaudit.vercel.app/registry/challenge)

# Step 2: Solve it (your agent solves the 3 steps)
TOKEN=$(curl -s -X POST https://skillaudit.vercel.app/registry/verify-challenge \
  -H 'Content-Type: application/json' \
  -d '{"challenge_id":"...","solutions":{...}}' | jq -r .registration_token)

# Step 3: Generate key
curl -s -X POST https://skillaudit.vercel.app/keys \
  -H 'Content-Type: application/json' \
  -d "{\"registration_token\":\"$TOKEN\",\"label\":\"my-agent\"}"
